terraform {
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
    null = {
      source  = "hashicorp/null"
      version = "~> 3.0"
    }
  }
}

provider "aws" {
  region = var.aws_region
}

variable "aws_region" {
  type    = string
  default = "us-east-1"
}

# Map of Instance IDs to their target instance types
variable "instances" {
  type = map(string)
  default = {
    "i-0123456789abcdef0" = "t3.medium"
    "i-0abcdef1234567890" = "t3.large"
  }
  description = "Map of instance_id => new_instance_type"
}

# Step 1: Stop all target instances in parallel
resource "aws_ec2_instance_state" "stop" {
  for_each    = var.instances
  instance_id = each.key
  state       = "stopped"
}

# Step 2: Create AMIs concurrently once each corresponding instance is stopped
resource "aws_ami_from_instance" "backup_image" {
  for_each           = var.instances
  name               = "backup-image-${each.key}-${formatdate("YYYYMMDDhhmmss", timestamp())}"
  source_instance_id = each.key
  snapshot_without_reboot = true

  depends_on = [aws_ec2_instance_state.stop]
}

# Step 3, 4 & 5: Resize, Start, and Wait for 2/2 Health Checks per instance
resource "null_resource" "modify_and_start" {
  for_each = var.instances

  triggers = {
    ami_id        = aws_ami_from_instance.backup_image[each.key].id
    instance_type = each.value
  }

  # Modify the instance type
  provisioner "local-exec" {
    command = "aws ec2 modify-instance-attribute --region ${var.aws_region} --instance-id ${each.key} --instance-type '{\"Value\": \"${each.value}\"}'"
  }

  # Start the instance
  provisioner "local-exec" {
    command = "aws ec2 start-instances --region ${var.aws_region} --instance-ids ${each.key}"
  }

  # Wait for 2/2 status checks to pass and print success
  provisioner "local-exec" {
    command = <<EOT
      echo "Waiting for 2/2 system checks to pass on ${each.key}..."
      aws ec2 wait instance-status-ok --region ${var.aws_region} --instance-ids ${each.key}
      echo "SUCCESS: Instance ${each.key} resized to ${each.value} and passed all health checks."
    EOT
  }

  depends_on = [aws_ami_from_instance.backup_image]
}

# Final Output Summary
output "execution_results" {
  value = {
    for id, type in var.instances :
    id => "Successfully resized to ${type} and verified healthy."
  }
  depends_on = [null_resource.modify_and_start]
}
